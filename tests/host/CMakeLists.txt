cmake_minimum_required(VERSION 3.20)
project(nrf54l15_host_tests C)
enable_testing()
if(NOT TARGET nrf54l15_core)
  add_subdirectory(../../lib/nrf54l15 nrf54l15)
endif()
add_executable(core_tests test_core.c)
target_link_libraries(core_tests PRIVATE nrf54l15_core)
if(MSVC)
  target_compile_options(nrf54l15_core PRIVATE /W4 /WX)
  target_compile_options(core_tests PRIVATE /W4 /WX)
else()
  target_compile_options(nrf54l15_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME core COMMAND core_tests)
